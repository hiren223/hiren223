```python
def build_model(data):

    data = understand(data)

    data = clean(data)

    insights = explore(data)

    features = engineer(insights)

    models = train_multiple_models(features)

    best_model = evaluate(
        models,
        metrics=[
            "precision",
            "recall",
            "f1_score",
```
